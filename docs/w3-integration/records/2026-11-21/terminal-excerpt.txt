import json
import re
import sys
from pathlib import Path

text = Path(sys.argv[1]).read_text()
pid = int(sys.argv[2])

marker = text.find('"type": "syscall_summary"')
assert marker >= 0, "缺少最终 JSON"

start = text.rfind("{", 0, marker)
report, _ = json.JSONDecoder().raw_decode(text[start:])

target = next((p for p in report["processes"] if p["pid"] == pid), None)
assert target is not None, "没有找到测试进程"

stats = {s["name"]: s for s in target["syscalls"]}
for name in ("read", "write"):
    item = stats.get(name)
    assert item is not None, f"缺少 {name} 统计"
    assert item["count"] >= 100, f"{name} 次数不足"
    assert item["error_count"] == 0, f"{name} 存在错误"
    print(name, json.dumps(item, ensure_ascii=False))

tracer = re.search(r"Tracer PID configured:\s*(\d+)", text)
assert tracer is not None, "缺少采集程序 PID"
assert all(p["pid"] != int(tracer.group(1)) for p in report["processes"]), \
    "采集程序自身未被过滤"

print("SYSCALL PASS")

assert len(lock_reports) == 1, "Lock 总结果缺失或数量异常"

reports = [
    ("CPU 全核压力", root / "cpu/full.json"),
    ("CPU 超额竞争", root / "cpu/contention.json"),
    ("Lock", lock_reports[0]),
    ("I/O", root / "io/results.json"),
]

failed = []
for name, path in reports:
    data = json.loads(path.read_text())
    passed = data.get("passed") is True
    print(f"{name}: {'PASS' if passed else 'FAIL'}")
    if not passed:
        failed.append(name)

print("结果目录：", root)
raise SystemExit(1 if failed else 0)
